 21# ==== Plot dispatch registry ====
 22# Populated by the graphics layer at import time. Keys are
 23# (_kind, dom, m); values are {"2d": cls, "3d": cls}.

 28def _dispatch(f, dim, **kwargs):
 29    """Instantiate the plot class registered for (f, dim)."""
 30    if not _PLOT_REGISTRY:
 31        raise ValueError(
 32            "plot dispatch table is empty — did you `import graphics`?"
 33        )
 34    key = f._shape_key()
 35    targets = _PLOT_REGISTRY.get(key)
 36    if not targets or targets.get(dim) is None:
 37        raise ValueError(f"{f._name} is not {dim}-plottable")
 38    return targets[dim](f, **kwargs)

 42def _duplicates(l):
 43    seen = set()

 54    Matrix-valued smooth functions $\\mathbb{R}^d \\rightarrow \\mathbb{R}^{m \\times n}$.

 58    def __init__(self, name, args, coords):
 59        """
 60        Constructs an object representing a function. Takes the arguments
 61
 62        - name: for the LaTeX display
 63
 64        - args: the argument vector as list of symbols
 65
 66        - coords: the coordinate functions as list of lists of symbolic expressions
 67        """
 68        if not isinstance(args, list):
 69            raise ValueError("Arguments have to be in a list, even if there is only one")
 70        if _duplicates(args):
 71            raise ValueError("Duplicate argument symbols")
 72        argset = set(args)
 73        cm = Matrix(coords)
 74        m, n = shape(cm)
 75        for i in range(m):
 76            for j in range(n):
 77                if not cm[i, j].free_symbols <= argset:
 78                    raise ValueError("Free variables in coordinate function")
 79        """Instance variables:"""
 80        self._name = name
 81        self._args = args
 82        self._coord_matrix = cm

108        """

124    def eval(self, *vals, precision=None):
125        """
126        Precondition: the number of arguments must be the same as the dimension of self's domain.
127
128        Returns the value of self evaluated on the arguments *vals. This value is symbolic by default and a float if 
129        precision is not $\\mathtt{None}$
130        """
131        if len(vals) != len(self.args):
132            raise ValueError("Wrong number of arguments")
133        sdict = dict(zip(self.args, vals))
134        m, n = shape(self.coord_matrix)
135        if (m, n) == (1, 1):
136            c = self.coord_matrix[0, 0].subs(sdict)
137            return c if precision is None else N(c, precision)
138        M = self.coord_matrix[:, :].subs(sdict)
139        return M if precision is None else N(M, precision)

141    def freeze(self, argvals, name=None):
142        """
143        Partial evaluation: fix some of self's arguments to values.
144
145        Preconditions:
146
147        (i) argvals is a list of pairs (symbol, value);
148
149        (ii) every symbol in argvals is among self's args, with no duplicates.
150
151        (iii) at least one argument of self is left free (freezing all arguments would give a point, not a function).
152
153        Returns a new Function in the remaining arguments. If name is
154        None, the name is auto-generated and renders in LaTeX as a restriction of self.
155        """
156        if not argvals:
157            raise ValueError("argvals must be a non-empty list of (symbol, value) pairs")
158        syms = [s for (s, _) in argvals]
159        if _duplicates(syms):
160            raise ValueError("Duplicate symbols in argvals")
161        if not set(syms) <= set(self.args):
162            raise ValueError("Frozen symbols must be among self's arguments")
163
164        newargs = [x for x in self.args if x not in set(syms)]
165        if not newargs:
166            raise ValueError("freeze() must leave at least one argument free")
167
168        if name is None:
169            frozen = ",".join(f"{latex(s)}={latex(v)}" for (s, v) in argvals)
170            name = f"{self._name}|_{{{frozen}}}"
171
172        sdict = dict(argvals)
173        M = self.coord_matrix[:, :].subs(sdict)
174        return Function(name, newargs, M)

178        """
179        Differential  
180        $$D_f: \\mathbb{R}^d \\rightarrow \\mathbb{R}^{m \\times d}$$ 
181
182        Precondition: self must be a vector-valued function $f: \\mathbb{R}^d \\rightarrow \\mathbb{R}^m$ for some

213        """

231    def _integrate_numeric(self, Phi, s_range, t_range):
232        import numpy as np
233        from scipy.integrate import dblquad
234        s, t = Phi.args
235        xs = Phi.coord_matrix[0, 0]
236        ys = Phi.coord_matrix[1, 0]
237        detJ = Phi.diff().coord_matrix.det()
238        integrand_expr = self.coord_matrix[0, 0].subs({self.args[0]: xs, self.args[1]: ys}) * detJ
239        f = lambdify((s, t), integrand_expr, "numpy")
240        def integrand(t, s):
241            return float(f(s, t))
242        result, _ = dblquad(integrand,s_range[0], s_range[1],lambda _: t_range[0], lambda _: t_range[1])
243        return result

245    def integrate(self, Phi, s_range, t_range, numeric=False):
246        """
247        Integral
248        $$
249            \\int_\\mathbf{D} \\, f \\, dx \\, dy = \\int_\\mathbf{D} f(\\Phi(s,t)) · det(D_{\\Phi}(s,t)) \\, dt \\, ds
250        $$
251
252        Preconditions: 
253        
254        $(i)$ self must be a scalar-valued function $f: \\mathbb{R}^2 \\rightarrow \\mathbb{R}$
255
256        $(ii)$ the parametrisation $\\Phi: \\mathbb{R}^2 \\rightarrow \\mathbb{R}^2$ of the domain of integration
257        must be *orientation-preserving*,
258        i.e. $det(D_{\\Phi}) \\geq 0$ on $\\mathbf{D} = [s_0,s_1] \\times [t_0,t_1]$. Notice that
259        given the precondition no absolute value is taken
260        (polar, cylindrical, spherical, and all the usual textbook parametrizations satisfy this on their standard 
261        domains).
262        
263        The value here is by default a SymPy expression (the exact integral, or an unevaluated
264        integral if SymPy cannot do it). **Warning**: performance can be a bottleneck, even if SymPy is capable
265        of providing an exact value. If `numeric = True`, the method returns a float.
266        """
267        if self._kind() != "scalar" or self.dom() != 2:
268            raise ValueError("integrate() needs a scalar function R^2 -> R")
269        if Phi.dom() != 2 or Phi.cod() != (2, 1):
270            raise ValueError("parametrization Phi must be R^2 -> R^2 (2 x 1)")
271
272        if numeric:
273            return self._integrate_numeric(Phi, s_range, t_range)
274
275        s, t = Phi.args
276        xs = Phi.coord_matrix[0, 0]
277        ys = Phi.coord_matrix[1, 0]
278        detJ = Phi.diff().coord_matrix.det()
279        integrand = self.coord_matrix[0, 0].subs({self.args[0]: xs, self.args[1]: ys}) * detJ
280        return sp_integrate(integrand,(s, s_range[0], s_range[1]),(t, t_range[0], t_range[1]))

282    def _line_integrand(self, gamma):
283        """
284        self: vector field
285        gamma: parametric curve
286
287        Common setup for integrate_along: returns (t, integrand_expr) where
288        integrand_expr is the scalar F(γ(t)) · γ'(t) in the parameter t.
289
290        Shared by the symbolic and numeric paths so they cannot diverge.
291        """
292        t = gamma.args[0]
293        subs = {fa: gc for fa, gc in zip(self.args, gamma.coord_matrix.col(0))}
294        gamma_prime = gamma.diff().coord_matrix.col(0)
295        integrand = sum(
296            F_i.subs(subs) * gp_i
297            for F_i, gp_i in zip(self.coord_matrix.col(0), gamma_prime)
298        )
299        return t, integrand

301    def integrate_along(self, gamma, t_range, numeric=False):
302        """
303            Line integral of a vector field:  
304            $$ \\int_\\gamma F dr = \\int_{t0}^{t1} F(\\gamma(t)) \\cdot \\gamma'(t) dt$$
305            
306    
307            Preconditions:
308            (i)  self  is a vector field $F: \\mathbb{R}^n \\rightarrow \\mathbb{R}^n$
309
310            (ii) gamma is a curve $\\gamma: \\mathbb{R} \\rightarrow \\mathbb{R}^n$ matching $F$'s dimension
311    
312            Orientation: reversing t_range reverses the sign, as expected.
313        """
314        n = self.dom()
315        if self._kind() != "vector" or self.cod() != (n, 1):
316            raise ValueError("integrate_along needs a vector field F: R^n -> R^n")
317        if gamma._kind() != "vector" or gamma.dom() != 1 or gamma.cod() != (n, 1):
318            raise ValueError("gamma must be a curve R -> R^n matching F's dimension")
319              
320        if numeric:
321            return self._integrate_along_numeric(gamma, t_range)
322        t, integrand = self._line_integrand(gamma)
323        return sp_integrate(expand(trigsimp(integrand)),(t, t_range[0], t_range[1]))

349    def integrate_over(self, S, u_range, v_range, numeric=False):
350        """
351        Flux of a vector field across a parametric surface:
352
353        $$\\int_S F \\cdot dS = \\int_{u_0}^{u_1} \\int_{v_0}^{v_1} F(S(u,v)) \\cdot (S_u \\times S_v) \\, dv \\, du$$
354
355        Preconditions:
356        (i)  self is a vector field $F : \\mathbb{R}^3 \\to \\mathbb{R}^3$;
357
358        (ii) S is a parametric surface $S : \\mathbb{R}^2 \\to \\mathbb{R}^3$;
359
360        (iii) the orientation of S (given by $S_u \\times S_v$) is the one
361                the user intends — reversing a parameter range reverses the sign.
362
363        The result may be negative; that is the orientation, not an error.
364        """
365        if self._kind() != "vector" or self.dom() != 3 or self.cod() != (3, 1):
366            raise ValueError("integrate_over needs a vector field F: R^3 -> R^3")
367        if S._kind() != "vector" or S.dom() != 2 or S.cod() != (3, 1):
368            raise ValueError("S must be a parametric surface R^2 -> R^3 (3 x 1)")
369
370        if numeric:
371            return self._integrate_over_numeric(S, u_range, v_range)
372
373        u, v, integrand = self._flux_integrand(S)
374        return sp_integrate(expand(trigsimp(integrand)),
375                            (u, u_range[0], u_range[1]),
376                            (v, v_range[0], v_range[1]))

378    def _flux_integrand(self, S):
379        """Returns (u, v, F(S(u,v)) · (S_u × S_v))."""
380        u, v = S.args
381        S_coords = S.coord_matrix.col(0)
382        subs = {fa: sc for fa, sc in zip(self.args, S_coords)}
383        S_u = [c.diff(u) for c in S_coords]
384        S_v = [c.diff(v) for c in S_coords]
385        cross = [S_u[1]*S_v[2] - S_u[2]*S_v[1],
386                S_u[2]*S_v[0] - S_u[0]*S_v[2],
387                S_u[0]*S_v[1] - S_u[1]*S_v[0]]
388        integrand = sum(F_i.subs(subs) * c_i
389                        for F_i, c_i in zip(self.coord_matrix.col(0), cross))
390        return u, v, integrand

392    def _surface_area_numeric(self, u_range, v_range, n=200):
393        import numpy as np
394        from scipy.integrate import dblquad
395
396        u, v = self.args
397        x_expr, y_expr, z_expr = (self.coord_matrix[i, 0] for i in range(3))
398        Phi_u = [x_expr.diff(u), y_expr.diff(u), z_expr.diff(u)]
399        Phi_v = [x_expr.diff(v), y_expr.diff(v), z_expr.diff(v)]
400
401        f_u = lambdify((u, v), Phi_u, "numpy")
402        f_v = lambdify((u, v), Phi_v, "numpy")
403
404        def area_elem(v, u):          # dblquad: inner u, outer v
405            pu = np.array(f_u(u, v))
406            pv = np.array(f_v(u, v))
407            return float(np.linalg.norm(np.cross(pu, pv)))
408
409        result, _ = dblquad(area_elem,u_range[0], u_range[1],   # inner: u
410                            lambda _: v_range[0], lambda _: v_range[1])  # outer: v
411        return result

413    def surface_area(self, u_range, v_range, numeric = True):
414        """
415        The surface area delimited by $f$ over some domain $\\mathbf{D} = [u_0,u_1] \\times [v_0,v_1]$, computed as
416        $$
417            \\int_{\\mathbf{D}} \\| \\Phi_u \\times \\Phi_v \\| \\, du \\, dv 
418        $$
419        where $\\Phi_u$ and $\\Phi_v$ are the partial derivatives of the coordinate functions.
420
421        Precondition: self must be a vector-valued function $f(u,v): \\mathbb{R}^2 \\rightarrow \\mathbb{R}^3$ 
422        
423        The value here is by default a 
424        float. If `numeric=False` it is a SymPy expression
425        (exact, or an unevaluated integral if
426        SymPy cannot do it). **Warning**: performanace can be a 
427        bottleneck, even if SymPy is capable of providing an exact value. If precision is not $\\mathtt{None}$, the method returns a float. 
428
429        Notice that orientation is not an issue here, since a norm is positive.
430        """
431        if self._kind() != "vector" or self.dom() != 2 or self.cod() != (3, 1):
432            raise ValueError("surface_area() needs a parametric surface R^2 -> R^3 (3 x 1)")
433
434        if numeric:
435            return self._surface_area_numeric(u_range, v_range)
436
437        u, v = self.args
438        x_expr = self.coord_matrix[0, 0]
439        y_expr = self.coord_matrix[1, 0]
440        z_expr = self.coord_matrix[2, 0]
441
442        Phi_u = Matrix([x_expr.diff(u), y_expr.diff(u), z_expr.diff(u)])
443        Phi_v = Matrix([x_expr.diff(v), y_expr.diff(v), z_expr.diff(v)])
444        area_elem = expand(trigsimp(Phi_u.cross(Phi_v).norm()))
445
446        return sp_integrate(area_elem, (u, u_range[0], u_range[1]),(v, v_range[0], v_range[1]))

449    def volume(self, r_range, s_range, t_range, numeric=True):
450        """
451        The volume delimited by $f$ overd $\\mathbf{D}$, computed as
452        $$
453            \\int_\\mathbf{D} det(D_f) \\, dr \\, ds \\, dt
454        $$
455
456        Precondition: self must be a vector-valued function $f(r,s,t): \\mathbb{R}^3 \\rightarrow \\mathbb{R}^3$

458
459        Notice that unlike what happens in the case of a surface area where an explicit domain parametrisation must be provided, in the case of 
460        a volume the parametrisation in question is the function itself.
461        
462        The value here is by default a float. If `numeric = False` it is a
463        SymPy expression (exact, or an unevaluated Integral if SymPy cannot do it). 
464        **Warning**: performanace can be a 
465        bottleneck, even if SymPy is capable of providing an exact value.

468            raise ValueError("volume() needs a parametric solid R^3 -> R^3 (3 x 1)")

493    def _flux_across_integrand(self, gamma):
494        """Returns (t, F_x·γ'_y − F_y·γ'_x), the integrand of ∮_C F·n ds."""
495        t = gamma.args[0]
496        subs = {fa: gc for fa, gc in zip(self.args, gamma.coord_matrix.col(0))}
497        gx = gamma.coord_matrix[0, 0]
498        gy = gamma.coord_matrix[1, 0]

503    def flux_across(self, gamma, t_range, numeric=False):
504        """
505        Flux of a plane vector field across a curve (2D divergence form of Green):
506
507        $$\\oint_C F \\cdot n \\, ds = \\int_{t_0}^{t_1} F(\\gamma(t)) \\cdot n \\, ||\\gamma'(t)|| \\, dt$$
508
509        where $n = (\\gamma'_y, -\\gamma'_x)/||\\gamma'||$ is the outward normal for
510        counterclockwise traversal. The $||\\gamma'||$ cancels, so the integrand is
511        $F_x \\gamma'_y - F_y \\gamma'_x$.
512
513        Orientation: reversing t_range reverses the sign.
514        """
515        if self._kind() != "vector" or self.dom() != 2 or self.cod() != (2, 1):
516            raise ValueError("flux_across needs a vector field F: R^2 -> R^2")
517        if gamma._kind() != "vector" or gamma.dom() != 1 or gamma.cod() != (2, 1):
518            raise ValueError("gamma must be a curve R -> R^2")
519
520        if numeric:
521            return self._flux_across_numeric(gamma, t_range)
522
523        t, integrand = self._flux_across_integrand(gamma)
524        return sp_integrate(expand(trigsimp(integrand)), (t, t_range[0], t_range[1]))

534    def div(self):
535        """
536        Divergence of a vector field:
537
538        $$\\nabla \\cdot F = \\sum_{i=1}^{n} \\frac{\\partial F_i}{\\partial x_i}$$
539
540        Precondition: self is a vector field $F: \\mathbb{R}^n \\to \\mathbb{R}^n$.

551        """
552        Curl of a plane vector field, in its scalar form:
553
554        $$\\operatorname{curl} F = \\frac{\\partial F_2}{\\partial x} - \\frac{\\partial F_1}{\\partial y}$$
555
556        Precondition: self is a vector field $F: \\mathbb{R}^2 \\to \\mathbb{R}^2$.
557
558        Returns a scalar function $\\mathbb{R}^2 \\to \\mathbb{R}$ (a new Function).
559
560        Note: in 2D the curl of a vector field is a scalar, not a vector — it is
561        the $z$-component of the 3D curl of $(F_1, F_2, 0)$. The 3D vector-valued
562        curl is a separate method (`curl3d`) to come.
563        """

568        x1, x2 = self.args
569        expr = Q.diff(x1) - P.diff(x2)

142    def freeze(self, argvals, name=None):
143        """
144        Partial evaluation: fix some of self's arguments to values.
145
146        Preconditions:
147
148        (i) argvals is a list of pairs (symbol, value);

286        gamma: parametric curve

350    def integrate_over(self, S, u_range, v_range, numeric=False):
351        """
352        Flux of a vector field across a parametric surface:
353
354        $$\\int_S F \\cdot dS = \\int_{u_0}^{u_1} \\int_{v_0}^{v_1} F(S(u,v)) \\cdot (S_u \\times S_v) \\, dv \\, du$$
355

357        (i)  self is a vector field $F : \\mathbb{R}^3 \\to \\mathbb{R}^3$;
358
359        (ii) S is a parametric surface $S : \\mathbb{R}^2 \\to \\mathbb{R}^3$;
360

362                the user intends — reversing a parameter range reverses the sign.
363
364        The result may be negative; that is the orientation, not an error.
365        """
366        if self._kind() != "vector" or self.dom() != 3 or self.cod() != (3, 1):
367            raise ValueError("integrate_over needs a vector field F: R^3 -> R^3")
368        if S._kind() != "vector" or S.dom() != 2 or S.cod() != (3, 1):
369            raise ValueError("S must be a parametric surface R^2 -> R^3 (3 x 1)")
370
371        if numeric:
372            return self._integrate_over_numeric(S, u_range, v_range)
373
374        u, v, integrand = self._flux_integrand(S)
375        return sp_integrate(expand(trigsimp(integrand)),
376                            (u, u_range[0], u_range[1]),
377                            (v, v_range[0], v_range[1]))

379    def _flux_integrand(self, S):
380        """Returns (u, v, F(S(u,v)) · (S_u × S_v))."""
381        u, v = S.args
382        S_coords = S.coord_matrix.col(0)
383        subs = {fa: sc for fa, sc in zip(self.args, S_coords)}
384        S_u = [c.diff(u) for c in S_coords]
385        S_v = [c.diff(v) for c in S_coords]
386        cross = [S_u[1]*S_v[2] - S_u[2]*S_v[1],
387                S_u[2]*S_v[0] - S_u[0]*S_v[2],
388                S_u[0]*S_v[1] - S_u[1]*S_v[0]]
389        integrand = sum(F_i.subs(subs) * c_i
390                        for F_i, c_i in zip(self.coord_matrix.col(0), cross))
391        return u, v, integrand

393    def _surface_area_numeric(self, u_range, v_range, n=200):
394        import numpy as np
395        from scipy.integrate import dblquad
396
397        u, v = self.args
398        x_expr, y_expr, z_expr = (self.coord_matrix[i, 0] for i in range(3))

400        Phi_v = [x_expr.diff(v), y_expr.diff(v), z_expr.diff(v)]
401
402        f_u = lambdify((u, v), Phi_u, "numpy")

405        def area_elem(v, u):          # dblquad: inner u, outer v
406            pu = np.array(f_u(u, v))

408            return float(np.linalg.norm(np.cross(pu, pv)))

411                            lambda _: v_range[0], lambda _: v_range[1])  # outer: v
