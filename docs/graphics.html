128    def _update_box_aspect(self):
129        if self._explicit_box_aspect:
130            return
131        x0, x1 = self.ax.get_xlim()
132        y0, y1 = self.ax.get_ylim()
133        z0, z1 = self.ax.get_zlim()
134        dx, dy, dz = x1 - x0, y1 - y0, z1 - z0
135        m = max(dx, dy, dz)
136        if m > 0:
137            eps = 1e-2
138            self.ax.set_box_aspect((max(dx/m, eps), max(dy/m, eps), max(dz/m, eps)))

26class Surface(Renderable):
27    """
28    Graph of a scalar function of two variables:  z = f(x, y).
29
30    Parameters
31    ----------
32    f : Function-like
33        Must satisfy _kind() == "scalar" and dom() == 2.
34    x_range, y_range : (float, float)
35    n : int
36        Grid resolution.
37    cmap : str or Colormap, optional
38    """
39
40    def __init__(self, f, x_range=(-10, 10), y_range=(-10, 10),n=100, cmap=None, color=None, alpha=None):
41        _check_function(f)
42        if f._kind() != "scalar" or f.dom() != 2:
43            raise ValueError("Surface needs a function R^2 -> R (scalar, dom 2)")
44        self.f = f
45        self.cmap = cmap
46        self.x_range = x_range
47        self.y_range = y_range
48        self.n = n
49        self.alpha = alpha
50        self.color = color
51        xs = _linspace(x_range, n)
52        ys = _linspace(y_range, n)
53        self.X, self.Y = np.meshgrid(xs, ys)
54        funcs = _numeric(f)          # scalar f: returns a 1-element list
55        self.Z = funcs(self.X, self.Y)[0]
56
57    def render(self, ax, **kw):
58        kw.setdefault("cmap", self.cmap)
59        kw.setdefault("alpha", self.alpha)
60        kw.setdefault("linewidth", 0)
61        kw.setdefault("antialiased", True)
62
63        if self.cmap is not None:
64            kw.setdefault("cmap", self.cmap)
65            kw.setdefault("vmin", self.Z.min())
66            kw.setdefault("vmax", self.Z.max())
67            kw.pop("color", None)          # cmap wins; drop any color kwarg
68        elif self.color is not None:
69            kw.setdefault("color", self.color)
70

128        norm = np.sqrt(nx*nx + ny*ny + nz*nz)
129        if norm == 0:
130            raise ValueError("Plane normal must be nonzero")

132        nvec = np.array([nx, ny, nz])
133

135        if abs(nx) < 0.9:

139        u = np.cross(nvec, helper)
140        u /= np.linalg.norm(u)
141        v = np.cross(nvec, u)

143        s = np.linspace(-size/2, size/2, max(n, 2))
144        t = np.linspace(-size/2, size/2, max(n, 2))
