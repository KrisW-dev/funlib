125    def eval(self, *vals, precision=None):
126        """
127        Precondition: the number of arguments must be the same as the dimension of self's domain.
128
129        Returns the value of self evaluated on the arguments *vals. This value is symbolic by default and a float if 
130        precision is not $\\mathtt{None}$

139        M = self.coord_matrix[:, :].subs(sdict)

142    def freeze(self, argvals, name=None):
143        """
144        Partial evaluation: fix some of self's arguments to values.
145
146        Preconditions:
147
148        (i) argvals is a list of pairs (symbol, value);
149
150        (ii) every symbol in argvals is among self's args, with no duplicates.
151
152        (iii) at least one argument of self is left free (freezing all arguments would give a point, not a function).
153
154        Returns a new Function in the remaining arguments. If name is
155        None, the name is auto-generated and renders in LaTeX as a restriction of self.
156        """
157        if not argvals:
158            raise ValueError("argvals must be a non-empty list of (symbol, value) pairs")
159        syms = [s for (s, _) in argvals]
160        if _duplicates(syms):
161            raise ValueError("Duplicate symbols in argvals")
162        if not set(syms) <= set(self.args):
163            raise ValueError("Frozen symbols must be among self's arguments")
164
165        newargs = [x for x in self.args if x not in set(syms)]
166        if not newargs:
167            raise ValueError("freeze() must leave at least one argument free")
168
169        if name is None:
170            frozen = ",".join(f"{latex(s)}={latex(v)}" for (s, v) in argvals)
171            name = f"{self._name}|_{{{frozen}}}"
172
173        sdict = dict(argvals)
174        M = self.coord_matrix[:, :].subs(sdict)
175        return Function(name, newargs, M)

246    def integrate(self, Phi, s_range, t_range, numeric=False):
247        """
248        Integral
249        $$
250            \\int_\\mathbf{D} \\, f \\, dx \\, dy = \\int_\\mathbf{D} f(\\Phi(s,t)) · det(D_{\\Phi}(s,t)) \\, dt \\, ds
251        $$
252
253        Preconditions: 
254        
255        $(i)$ self must be a scalar-valued function $f: \\mathbb{R}^2 \\rightarrow \\mathbb{R}$
256
257        $(ii)$ the parametrisation $\\Phi: \\mathbb{R}^2 \\rightarrow \\mathbb{R}^2$ of the domain of integration
258        must be *orientation-preserving*,
259        i.e. $det(D_{\\Phi}) \\geq 0$ on $\\mathbf{D} = [s_0,s_1] \\times [t_0,t_1]$. Notice that
260        given the precondition no absolute value is taken
261        (polar, cylindrical, spherical, and all the usual textbook parametrizations satisfy this on their standard 
262        domains).
263        
264        The value here is by default a SymPy expression (the exact integral, or an unevaluated
265        integral if SymPy cannot do it). **Warning**: performance can be a bottleneck, even if SymPy is capable
266        of providing an exact value. If `numeric = True`, the method returns a float.
267        """
268        if self._kind() != "scalar" or self.dom() != 2:
269            raise ValueError("integrate() needs a scalar function R^2 -> R")
270        if Phi.dom() != 2 or Phi.cod() != (2, 1):
271            raise ValueError("parametrization Phi must be R^2 -> R^2 (2 x 1)")
272
273        if numeric:
274            return self._integrate_numeric(Phi, s_range, t_range)
275
276        s, t = Phi.args
277        xs = Phi.coord_matrix[0, 0]
278        ys = Phi.coord_matrix[1, 0]
279        detJ = Phi.diff().coord_matrix.det()
280        integrand = self.coord_matrix[0, 0].subs({self.args[0]: xs, self.args[1]: ys}) * detJ
281        return sp_integrate(integrand,(s, s_range[0], s_range[1]),(t, t_range[0], t_range[1]))

283    def _line_integrand(self, gamma):
284        """
285        self: vector field
286        gamma: parametric curve
287
288        Common setup for integrate_along: returns (t, integrand_expr) where
289        integrand_expr is the scalar F(γ(t)) · γ'(t) in the parameter t.
290
291        Shared by the symbolic and numeric paths so they cannot diverge.
292        """
293        t = gamma.args[0]
294        subs = {fa: gc for fa, gc in zip(self.args, gamma.coord_matrix.col(0))}
295        gamma_prime = gamma.diff().coord_matrix.col(0)
296        integrand = sum(
297            F_i.subs(subs) * gp_i
298            for F_i, gp_i in zip(self.coord_matrix.col(0), gamma_prime)
299        )
300        return t, integrand

307

350    def integrate_over(self, S, u_range, v_range, numeric=False):
351        """
352        Flux of a vector field across a parametric surface:
353
354        $$\\int_S F \\cdot dS = \\int_{u_0}^{u_1} \\int_{v_0}^{v_1} F(S(u,v)) \\cdot (S_u \\times S_v) \\, dv \\, du$$
355
356        Preconditions:
357        (i)  self is a vector field $F : \\mathbb{R}^3 \\to \\mathbb{R}^3$;
358
359        (ii) S is a parametric surface $S : \\mathbb{R}^2 \\to \\mathbb{R}^3$;
360
361        (iii) the orientation of S (given by $S_u \\times S_v$) is the one
362                the user intends — reversing a parameter range reverses the sign.
363
364        The result may be negative; that is the orientation, not an error.
365        """
366        if self._kind() != "vector" or self.dom() != 3 or self.cod() != (3, 1):
367            raise ValueError("integrate_over needs a vector field F: R^3 -> R^3")
368        if S._kind() != "vector" or S.dom() != 2 or S.cod() != (3, 1):
369            raise ValueError("S must be a parametric surface R^2 -> R^3 (3 x 1)")
370
371        if numeric:
372            return self._integrate_over_numeric(S, u_range, v_range)
373
374        u, v, integrand = self._flux_integrand(S)
375        return sp_integrate(expand(trigsimp(integrand)),
376                            (u, u_range[0], u_range[1]),
377                            (v, v_range[0], v_range[1]))

379    def _flux_integrand(self, S):
380        """Returns (u, v, F(S(u,v)) · (S_u × S_v))."""
381        u, v = S.args
382        S_coords = S.coord_matrix.col(0)
383        subs = {fa: sc for fa, sc in zip(self.args, S_coords)}
384        S_u = [c.diff(u) for c in S_coords]
385        S_v = [c.diff(v) for c in S_coords]
386        cross = [S_u[1]*S_v[2] - S_u[2]*S_v[1],
387                S_u[2]*S_v[0] - S_u[0]*S_v[2],
388                S_u[0]*S_v[1] - S_u[1]*S_v[0]]
389        integrand = sum(F_i.subs(subs) * c_i
390                        for F_i, c_i in zip(self.coord_matrix.col(0), cross))
391        return u, v, integrand

393    def _surface_area_numeric(self, u_range, v_range, n=200):
394        import numpy as np
395        from scipy.integrate import dblquad
396
397        u, v = self.args
398        x_expr, y_expr, z_expr = (self.coord_matrix[i, 0] for i in range(3))
399        Phi_u = [x_expr.diff(u), y_expr.diff(u), z_expr.diff(u)]
400        Phi_v = [x_expr.diff(v), y_expr.diff(v), z_expr.diff(v)]
401
402        f_u = lambdify((u, v), Phi_u, "numpy")
403        f_v = lambdify((u, v), Phi_v, "numpy")
404
405        def area_elem(v, u):          # dblquad: inner u, outer v
406            pu = np.array(f_u(u, v))
407            pv = np.array(f_v(u, v))
408            return float(np.linalg.norm(np.cross(pu, pv)))
409
410        result, _ = dblquad(area_elem,u_range[0], u_range[1],   # inner: u
411                            lambda _: v_range[0], lambda _: v_range[1])  # outer: v
412        return result

504    def flux_across(self, gamma, t_range, numeric=False):

519            raise ValueError("gamma must be a curve R -> R^2")

522            return self._flux_across_numeric(gamma, t_range)
523
524        t, integrand = self._flux_across_integrand(gamma)
525        return sp_integrate(expand(trigsimp(integrand)), (t, t_range[0], t_range[1]))
